import serial
import time

from serial.tools.list_ports import comports
DEV = "/dev/ttyUSB0"
comm = serial.Serial(DEV, 115200, timeout=10)

START_BIT = b'\x68'
WATTMETER_ID = [hex(x) for x in bytes.fromhex('010097796151')]

def _gen_cmd(address: str, control_code: bytes, data: bytes):
  assert len(address) == 12, "Address must be 12 hex digits"
  data_length = len(data)
  cmd = START_BIT + bytes.fromhex(address) + START_BIT + control_code + int.to_bytes(data_length) + data

  checksum = sum(cmd) & 0xFF
  cmd = cmd + int.to_bytes(checksum & 0xFF) + b'\x16'
  return cmd

cmd = _gen_cmd('010000000000', b'\x01', b'\x00\x00\x00\x00')

**03 TCP 抄表**

```python
import socket
import time

##** 基础配置
UDP_IP = "192.168.10.73"
UDP_PORT = 8899

START_BIT = 0x68
END_BIT = 0x16
SENDER_PROCESS = 0x33

##** 指令生成
def _gen_cmd(address: str, control_code: int, data: list[int]):
  assert len(address) == 12, "Address must be 12 hex digits"
  cmd = [START_BIT]
  WATTMETER_ID = [x for x in bytes.fromhex(address)
  WATTMETER_ID.reverse()

  data_length = len(data)
  cmd.extend(WATTMETER_ID)
  cmd.append(START_BIT)
  cmd.append(control_code)
  cmd.append(data_length)
  for d in data:
    cmd.append((d + SENDER_PROCESS) & 0xFF)

  checksum = sum(cmd) & 0xFF
  cmd.append(checksum & 0xFF)
  cmd.append(END_BIT)
  return bytes(cmd)

##** 抄表
cmd = _gen_cmd('591820013440', 0x11, [0x00, 0x00, 0x00, 0x00])

##** 打印信息
print("Command:", cmd.hex(' '))

print("UDP target IP: %s" % UDP_IP)
print("UDP target port: %s" % UDP_PORT)

##** 发送
sock = socket.socket(socket.AF_INET,     # Internet
                     socket.SOCK_STREAM) # TCP
sock.connect((UDP_IP, UDP_PORT))
# sock.setblocking = False
sent = sock.send(cmd)
sock.setblocking(False)
##** 接收
while(1):
  receive = b''
  try:
    receive = sock.recv(255)
  except socket.error as e:
      if e:
          print("No data available yet (non-blocking)")
      else:
          print(f"Receive error: {e}")
  print(receive.hex(' '))
  # fe fe fe 68 40 34 01 20 18 59 68 91 08 33 33 33 33 c4 53 7a 34 00 16
  # fe fe fe 68 40 34 01 20 18 59 68 d1 01 35 dd 16
  time.sleep(1)
sock.close()
```
