# 有人物联网串口配置

**01 配置代码**

```python
import serial
import time

from serial.tools.list_ports import comports
DEV = "/dev/ttyUSB0"
comm = serial.Serial(DEV, 115200, timeout=10)
COMMANDS = [
  b'AT+WMODE=STA\x0D',
  b'AT+WSSSID=_box\x0D',
  # b'AT+WSSSID',
  b'AT+WSKEY=WPA2PSK,AES,Wifimima8nengwei0\x0D',
  # b'AT+ASWD',
  b'AT+Z\x0D',
]

comm.write(b'+++')
print(comm.read_until(b'a'))
comm.write(b'a')
print(comm.read_until(b'+ok\r\n\r\n'))

for cmd in COMMANDS:
  comm.write(cmd)
  time.sleep(1)

  print("send:", cmd)
  try:
    print(comm.read_until(b'\r\n\r\n'))
  except Exception as e:
    print(e)
    print("reopen serial port.")
    comm = serial.Serial(DEV, 115200, timeout=10)
```

**02 电表通信代码**

```python
import serial
import time

from serial.tools.list_ports import comports
DEV = "/dev/ttyUSB0"
comm = serial.Serial(DEV, 115200, timeout=10)

START_BIT = b'\x68'
WATTMETER_ID = [hex(x) for x in bytes.fromhex('010097796151')]

def _gen_cmd(address: str, control_code: bytes, data: bytes):
  assert len(address) == 12, "Address must be 12 hex digits"
  data_length = len(data)
  cmd = START_BIT + bytes.fromhex(address) + START_BIT + control_code + int.to_bytes(data_length) + data

  checksum = sum(cmd) & 0xFF
  cmd = cmd + int.to_bytes(checksum & 0xFF) + b'\x16'
  return cmd

cmd = _gen_cmd('010000000000', b'\x01', b'\x00\x00\x00\x00')
print("Command:", cmd.hex(' '))
exit(0)

comm.write(cmd)
time.sleep(1)

print("send:", cmd)
try:
  print(comm.read_until(b'\r\n\r\n'))
except Exception as e:
  print(e)
  print("reopen serial port.")
  comm = serial.Serial(DEV, 115200, timeout=10)

```



### 电表需求

**01 [DL/T 645-2007](https://github.com/AuroraRAS/electricMeterReader/blob/master/DLT645-2007.pdf)**

- 【P20】（当前）组合有功总电能
- 【P42】瞬时有功功率
- 【P44】零线电流
